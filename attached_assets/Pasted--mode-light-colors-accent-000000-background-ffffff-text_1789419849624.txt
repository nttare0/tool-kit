  <script>
    async function fetchSvglLogo(name) {
      try {
        const response = await fetch(`https://api.svgl.app?search=${encodeURIComponent(name)}`);
        const data = await response.json();

        if (data && data.length > 0) {
          const match = data[0];
          // Handle 'route' being either a string URL or a ThemeOptions object
          if (typeof match.route === 'string') {
            return match.route;
          } else if (match.route && match.route.light) {
            return match.route.light;
          }
        }
      } catch (error) {
        console.error("Error fetching logo from SVGL:", error);
      }
      return null;
    }

    async function addTool() {
      const name = document.getElementById('toolName').value.trim();
      const url = document.getElementById('toolUrl').value.trim();
      const desc = document.getElementById('toolDesc').value.trim();

      if (!name || !url) {
        alert("Please provide both a name and a valid URL!");
        return;
      }

      // 1. Screenshot Preview API
      const screenshotApiUrl = `https://api.microlink.io/?url=${encodeURIComponent(url)}&screenshot=true&meta=false&embed=screenshot.url`;

      // 2. Fetch Logo from SVGL API with Fallback to Google Favicon API
      const svglLogoUrl = await fetchSvglLogo(name);
      const fallbackFavicon = `https://www.google.com/s2/favicons?domain=${encodeURIComponent(url)}&sz=64`;
      const finalLogoUrl = svglLogoUrl || fallbackFavicon;

      // 3. Render Card
      const card = document.createElement('div');
      card.className = 'card';
      card.innerHTML = `
        <img class="preview-img" src="${screenshotApiUrl}" alt="${name} Homepage Preview" loading="lazy">
        <div class="card-content">
          <div class="card-header">
            <img class="brand-logo" src="${finalLogoUrl}" alt="${name} Logo" onerror="this.src='${fallbackFavicon}'">
            <h3>${name}</h3>
          </div>
          <p>${desc}</p>
          <a href="${url}" target="_blank">Visit Tool</a>
        </div>
      `;

      document.getElementById('toolGrid').appendChild(card);

      // Clear Form
      document.getElementById('toolName').value = '';
      document.getElementById('toolUrl').value = '';
      document.getElementById('toolDesc').value = '';
    }
  </script>
