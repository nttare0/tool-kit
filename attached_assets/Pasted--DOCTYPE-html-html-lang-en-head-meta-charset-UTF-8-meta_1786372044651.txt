    chipsWrap.querySelectorAll('.chip').forEach(btn=>{
      btn.addEventListener('click', ()=>{
        activeCategory = btn.dataset.cat;
        renderChips();
        renderGrid();
      });
    });
  }

  function renderGrid(){
    const term = searchTerm.trim().toLowerCase();
    const filtered = allTools.filter(t=>{
      const matchesCat = activeCategory === "All" || t.cat === activeCategory;
      const matchesTerm = !term ||
        t.name.toLowerCase().includes(term) ||
        t.desc.toLowerCase().includes(term) ||
        t.cat.toLowerCase().includes(term);
      return matchesCat && matchesTerm;
    });

    status.textContent = `${filtered.length} tool${filtered.length===1?'':'s'} on the board${activeCategory!=='All' ? ' · '+activeCategory : ''}`;

    if(filtered.length === 0){
      grid.innerHTML = `<div class="empty" style="grid-column:1/-1">
        <strong>Nothing here yet</strong>
        Try a different search, or be the first to hang a tool in this category.
      </div>`;
      return;
    }

    grid.innerHTML = filtered.map(t=>{
      const isFav = favorites.includes(t.id);
      return `
      <a class="card" href="${escapeHtml(normalizeUrl(t.url))}" target="_blank" rel="noopener noreferrer" data-id="${t.id}">
        <div class="peg"></div>
        <div class="card-top">
          <div class="card-name">${escapeHtml(t.name)}</div>
          <button class="fav-btn ${isFav?'active':''}" data-id="${t.id}" title="Pin as favorite" type="button">${isFav ? '★' : '☆'}</button>
        </div>
        <div class="card-desc">${escapeHtml(t.desc)}</div>
        <div class="card-bottom">
          <span class="tag">${escapeHtml(t.cat)}</span>
          <span class="open-arrow">open ↗</span>
        </div>
      </a>`;
    }).join('');

    grid.querySelectorAll('.fav-btn').forEach(btn=>{
      btn.addEventListener('click', (e)=>{
        e.preventDefault();
        e.stopPropagation();
        toggleFavorite(btn.dataset.id);
      });
    });
  }

  async function toggleFavorite(id){
    if(favorites.includes(id)){
      favorites = favorites.filter(f=>f!==id);
    } else {
      favorites.push(id);
    }
    renderGrid();
    if(storageOK){
      try{
        await window.storage.set('favorites', JSON.stringify(favorites), false);
      }catch(e){ console.error('Could not save favorites', e); }
    }
  }

  async function loadFavorites(){
    try{
      const res = await window.storage.get('favorites', false);
      if(res && res.value) favorites = JSON.parse(res.value);
    }catch(e){
      favorites = [];
    }
  }

  async function loadTools(){
    status.textContent = "loading the board…";
    try{
      const listRes = await window.storage.list('tool:', true);
      const keys = (listRes && listRes.keys) ? listRes.keys : [];

      if(keys.length === 0){
        // Seed the board for the first time
        for(const t of SEED){
          const id = 'seed-' + t.name.toLowerCase().replace(/[^a-z0-9]+/g,'-');
          const record = { id, ...t };
          try{
            await window.storage.set('tool:' + id, JSON.stringify(record), true);
          }catch(e){ console.error('Seed failed for', t.name, e); }
        }
        allTools = SEED.map(t => ({ id: 'seed-' + t.name.toLowerCase().replace(/[^a-z0-9]+/g,'-'), ...t }));
      } else {
        const results = await Promise.all(keys.map(async k=>{
          try{
            const r = await window.storage.get(k, true);
            return r && r.value ? JSON.parse(r.value) : null;
          }catch(e){ return null; }
        }));
        allTools = results.filter(Boolean).sort((a,b)=> (b.createdAt||0) - (a.createdAt||0));
      }
    }catch(e){
      console.error('Storage unavailable, using local seed only', e);
      storageOK = false;
      allTools = SEED.map(t => ({ id: 'local-' + t.name.toLowerCase().replace(/[^a-z0-9]+/g,'-'), ...t }));
    }
    renderGrid();
  }

  async function addTool(record){
    const key = 'tool:' + record.id;
    if(!storageOK){
      allTools.unshift(record);
      renderGrid();
      return true;
    }
    try{
      const result = await window.storage.set(key, JSON.stringify(record), true);
      if(!result) return false;
      allTools.unshift(record);
      renderGrid();
      return true;
    }catch(e){
      console.error('Failed to save tool', e);
      return false;
    }
  }

  // ---- events ----
  searchInput.addEventListener('input', (e)=>{
    searchTerm = e.target.value;
    renderGrid();
  });

  document.getElementById('openAdd').addEventListener('click', ()=>{
    overlay.classList.add('open');
    formMsg.className = 'form-msg';
    formMsg.textContent = '';
    document.getElementById('f-name').focus();
  });
  function closeModal(){
    overlay.classList.remove('open');
    form.reset();
    formMsg.className = 'form-msg';
    formMsg.textContent = '';
  }
  document.getElementById('closeModal').addEventListener('click', closeModal);
  document.getElementById('cancelAdd').addEventListener('click', closeModal);
  overlay.addEventListener('click', (e)=>{ if(e.target === overlay) closeModal(); });
  document.addEventListener('keydown', (e)=>{ if(e.key === 'Escape' && overlay.classList.contains('open')) closeModal(); });

  form.addEventListener('submit', async (e)=>{
    e.preventDefault();
    const name = document.getElementById('f-name').value.trim();
    const url = document.getElementById('f-url').value.trim();
    const desc = document.getElementById('f-desc').value.trim();
    const cat = document.getElementById('f-cat').value;

    if(!name || !url || !desc){
      formMsg.className = 'form-msg error';
      formMsg.textContent = 'Fill in every field before hanging it up.';
      return;
    }

    submitBtn.disabled = true;
    submitBtn.textContent = 'Hanging…';

    const id = Date.now().toString(36) + Math.random().toString(36).slice(2,7);
    const record = { id, name, url: normalizeUrl(url), desc, cat, createdAt: Date.now() };

    const ok = await addTool(record);

    submitBtn.disabled = false;
    submitBtn.textContent = 'Hang it up';

    if(ok){
      formMsg.className = 'form-msg success';
      formMsg.textContent = 'Added to the board for everyone.';
      setTimeout(closeModal, 900);
    } else {
      formMsg.className = 'form-msg error';
      formMsg.textContent = 'Could not save — try again in a moment.';
    }
  });

  // ---- import bookmarks ----
  function resetImport(){
    parsedBookmarks = [];
    fileInput.value = '';
    fdLabel.innerHTML = 'Drop the exported <b>bookmarks.html</b> file here, or click to choose it';
    importList.className = 'import-list';
    importList.innerHTML = '';
    importControls.style.display = 'none';
    importBtn.disabled = true;
    importMsg.className = 'form-msg';
    importMsg.textContent = '';
  }

  function openImportModal(){
    resetImport();
    importOverlay.classList.add('open');
  }
  function closeImportModal(){
    importOverlay.classList.remove('open');
  }

  document.getElementById('openImport').addEventListener('click', openImportModal);
  document.getElementById('closeImport').addEventListener('click', closeImportModal);
  document.getElementById('cancelImport').addEventListener('click', closeImportModal);
  importOverlay.addEventListener('click', (e)=>{ if(e.target === importOverlay) closeImportModal(); });

  function parseBookmarksHtml(htmlText){
    const doc = new DOMParser().parseFromString(htmlText, 'text/html');
    const anchors = Array.from(doc.querySelectorAll('a[href]'));
    const seen = new Set();
    const out = [];
    for(const a of anchors){
      const href = (a.getAttribute('href') || '').trim();
      const title = (a.textContent || '').trim();
      if(!href || !/^https?:\/\//i.test(href)) continue;
      if(seen.has(href)) continue;
      seen.add(href);
      out.push({ name: title || href.replace(/^https?:\/\//,'').split('/')[0], url: href });
      if(out.length >= 500) break;
    }
    return out;
  }

  function renderImportList(){
    if(parsedBookmarks.length === 0){
      importList.className = 'import-list';
      importControls.style.display = 'none';
      importBtn.disabled = true;
      return;
    }
    importList.className = 'import-list open';
    importControls.style.display = 'flex';
    importList.innerHTML = parsedBookmarks.map((b,i)=>`
      <label class="import-row">
        <input type="checkbox" class="imp-check" data-i="${i}" checked>
        <div class="import-row-text">
          <div class="import-row-name">${escapeHtml(b.name)}</div>
          <div class="import-row-url">${escapeHtml(b.url)}</div>
        </div>
      </label>
    `).join('');
    updateImportBtnState();
    importList.querySelectorAll('.imp-check').forEach(cb=>{
      cb.addEventListener('change', updateImportBtnState);
    });
  }

  function updateImportBtnState(){
    const checked = importList.querySelectorAll('.imp-check:checked').length;
    importBtn.disabled = checked === 0;
    importBtn.textContent = checked > 0 ? `Import ${checked} tool${checked===1?'':'s'}` : 'Import selected';
  }

  document.getElementById('selectAll').addEventListener('click', ()=>{
    importList.querySelectorAll('.imp-check').forEach(cb=> cb.checked = true);
    updateImportBtnState();
  });
  document.getElementById('selectNone').addEventListener('click', ()=>{
    importList.querySelectorAll('.imp-check').forEach(cb=> cb.checked = false);
    updateImportBtnState();
  });

  fileInput.addEventListener('change', async ()=>{
    const file = fileInput.files && fileInput.files[0];
    if(!file) return;
    fdLabel.innerHTML = `Reading <b>${escapeHtml(file.name)}</b>…`;
    try{
      const text = await file.text();
      parsedBookmarks = parseBookmarksHtml(text);
      if(parsedBookmarks.length === 0){
        fdLabel.innerHTML = 'No links found in that file — export bookmarks as HTML and try again';
        importMsg.className = 'form-msg error';
        importMsg.textContent = 'Nothing to import.';
        return;
      }
      fdLabel.innerHTML = `Found <b>${parsedBookmarks.length}</b> bookmark${parsedBookmarks.length===1?'':'s'} — review below`;
      importMsg.className = 'form-msg';
      importMsg.textContent = '';
      renderImportList();
    }catch(e){
      console.error('Failed to read bookmarks file', e);
      fdLabel.innerHTML = 'Could not read that file — make sure it\'s the exported bookmarks HTML';
    }
  });

  importBtn.addEventListener('click', async ()=>{
    const checks = Array.from(importList.querySelectorAll('.imp-check:checked'));
    if(checks.length === 0) return;
    const cat = importCat.value;
    const existingUrls = new Set(allTools.map(t => t.url));

    importBtn.disabled = true;
    let done = 0, skipped = 0;
    const total = checks.length;

    for(const cb of checks){
      const b = parsedBookmarks[parseInt(cb.dataset.i, 10)];
      const url = normalizeUrl(b.url);
      if(existingUrls.has(url)){ skipped++; continue; }
      existingUrls.add(url);
      const id = Date.now().toString(36) + Math.random().toString(36).slice(2,7);
      const record = {
        id, name: b.name.slice(0,60) || url, url,
        desc: 'Imported from browser bookmarks.',
        cat, createdAt: Date.now()
      };
      await addTool(record);
      done++;
      importBtn.textContent = `Importing ${done + skipped}/${total}…`;
    }

    importMsg.className = 'form-msg success';
    importMsg.textContent = `Added ${done} tool${done===1?'':'s'} to the board${skipped ? `, skipped ${skipped} already there` : ''}.`;
    importBtn.textContent = 'Import selected';
    setTimeout(closeImportModal, 1400);
  });

  // ---- init ----
  renderChips();
  loadFavorites().then(loadTools);
})();
</script>
